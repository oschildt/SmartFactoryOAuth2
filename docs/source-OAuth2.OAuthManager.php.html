32:      * Each call replaces all settings. Failed initialization preserves the previous configuration.
33:      *

41:      * encryption_algorithm and secret_key or public_key). Collaborators are optional for offline
42:      * verification; RSA verification needs no private key. Issuer and audience restrictions are

203:     /**
204:      * Loads keys and binds their validated type and strength to one algorithm.
205:      *
206:      * @param array<string, mixed> $config
207:      * The algorithm and raw HMAC bytes or local RSA PEM paths.
208:      *
209:      * @param bool $allow_private
210:      * Whether a private signing key may be supplied.
211:      *
212:      * @return array{algorithm: string, verification: string|\OpenSSLAsymmetricKey, signing: string|\OpenSSLAsymmetricKey|null}
213:      * Returns parsed keys; signing is null for public-key-only verification.
214:      *
215:      * @throws AuthSystemException
216:      * If the algorithm, key material, password, key type, pair or length is invalid.
217:      *
218:      * @author Oleg Schildt
219:      */
220:     protected function prepareKey(array $config, bool $allow_private): array
221:     {
222:         $algorithm = $config['encryption_algorithm'] ?? null;
223:         $sizes = ['HS256' => 32, 'HS384' => 48, 'HS512' => 64, 'RS256' => 32, 'RS384' => 48, 'RS512' => 64];
224: 
225:         if (!is_string($algorithm) || !isset($sizes[$algorithm])) {
226:             throw new AuthSystemException('An unsupported signing algorithm was configured.', SmartException::ERR_CODE_CONFIG_ERROR);
227:         }
228: 
229:         if (str_starts_with($algorithm, 'HS')) {
230:             $secret = $config['secret_key'] ?? null;
231: 
232:             if (!is_string($secret) || strlen($secret) < $sizes[$algorithm]) {
233:                 throw new AuthSystemException('The HMAC key is shorter than the hash output.', SmartException::ERR_CODE_CONFIG_ERROR);
234:             }
235: 
236:             return ['algorithm' => $algorithm, 'verification' => $secret, 'signing' => $allow_private ? $secret : null];
237:         }
238: 
239:         $private = null;
240:         $public = null;
241:         $password = $config['pass_phrase'] ?? '';
242: 
243:         if (!is_string($password)) {
244:             throw new AuthSystemException('The private-key password must be a string.', SmartException::ERR_CODE_CONFIG_ERROR);
245:         }
246: 
247:         if ($allow_private && ($config['private_key'] ?? null) !== null) {
248:             $private = @openssl_pkey_get_private($this->readKeyFile($config['private_key']), $password);
249: 
250:             if ($private === false) {
251:                 throw new AuthSystemException('The private signing key could not be loaded.', SmartException::ERR_CODE_CONFIG_ERROR);
252:             }
253: 
254:             $details = openssl_pkey_get_details($private);
255:             $public = $details === false ? false : openssl_pkey_get_public($details['key']);
256:         }
257: 
258:         if (($config['public_key'] ?? null) !== null) {
259:             $public = @openssl_pkey_get_public($this->readKeyFile($config['public_key']));
260:         }
261: 
262:         if (!$public instanceof \OpenSSLAsymmetricKey) {
263:             throw new AuthSystemException('A valid RSA public verification key is required.', SmartException::ERR_CODE_CONFIG_ERROR);
264:         }
265: 
266:         $details = openssl_pkey_get_details($public);
267: 
268:         if ($details === false || $details['type'] !== OPENSSL_KEYTYPE_RSA || $details['bits'] < 2048) {
269:             throw new AuthSystemException('RSA keys must have type RSA and at least 2048 bits.', SmartException::ERR_CODE_CONFIG_ERROR);
270:         }
271: 
272:         if ($private !== null) {
273:             $private_details = openssl_pkey_get_details($private);
274: 
275:             if ($private_details === false || $private_details['type'] !== OPENSSL_KEYTYPE_RSA || $private_details['key'] !== $details['key']) {
276:                 throw new AuthSystemException('The RSA public and private keys do not form a pair.', SmartException::ERR_CODE_CONFIG_ERROR);
277:             }
278:         }
279: 
280:         return ['algorithm' => $algorithm, 'verification' => $public, 'signing' => $private];
281:     } // prepareKey

396:      * Signs a serialized JWT with the configured active key.
397:      *
398:      * @param string $input
399:      * The encoded header and payload separated by a period.
400:      *
401:      * @return string
402:      * Returns the binary HMAC or RSA signature.
403:      *
404:      * @throws AuthSystemException
405:      * If initialization, a private signing key or RSA signing is unavailable.
406:      *
407:      * @author Oleg Schildt
408:      */
409:     protected function generateJwtSignature(string $input): string
410:     {
411:         $this->validateParameters();
412: 
413:         $key = $this->configuration['active_key'];
414:         $algorithm = $key['algorithm'];
415:         $hash = 'sha' . substr($algorithm, 2);
416: 
417:         if ($key['signing'] === null) {
418:             throw new AuthSystemException('This manager has no private signing key.', SmartException::ERR_CODE_CONFIG_ERROR);
419:         }

422:             return hash_hmac($hash, $input, $key['signing'], true);
423:         }

426:

474:     /**
475:      * Parses a JWT and verifies its signature using locally configured keys only.
476:      *
477:      * @param ?string $jwt_access_token
478:      * A compact signed access token; null and empty strings are rejected.
479:      *
480:      * @param bool $verify_signature
481:      * Must be true; unverified extraction is deliberately unavailable.
482:      *
483:      * @return array<string, mixed>
484:      * Returns an object-shaped payload after successful signature verification.
485:      *
486:      * @throws MissingParametersException
487:      * If the JWT is null or empty.
488:      *
489:      * @throws InvalidTokenException
490:      * If encoding, JSON objects, header, key ID, algorithm or signature is invalid.
491:      *
492:      * @throws AuthSystemException
493:      * If the manager is uninitialized or cryptographic verification fails internally.
494:      *
495:      * @author Oleg Schildt
496:      */
497:     protected function getJwtPayload(?string $jwt_access_token, bool $verify_signature = true): array
498:     {
499:         $this->validateParameters();
500: 
501:         if ($jwt_access_token === null || $jwt_access_token === '') {
502:             throw new MissingParametersException('The access token is not specified.');
503:         }
504: 
505:         if (!$verify_signature || strlen($jwt_access_token) > 16384) {
506:             throw new InvalidTokenException('The JWT cannot be processed.');
507:         }
508: 
509:         $segments = explode('.', $jwt_access_token);
510: 
511:         if (count($segments) !== 3) {
512:             throw new InvalidTokenException('The JWT must contain exactly three segments.');
513:         }
514: 
515:         [$head, $body, $signature] = $segments;
516:         $header_json = $this->urlSafeB64Decode($head);
517:         $payload_json = $this->urlSafeB64Decode($body);
518:         $signature = $this->urlSafeB64Decode($signature);
519: 
520:         try {
521:             $header = json_decode($header_json, false, 64, JSON_THROW_ON_ERROR);
522:             $payload = json_decode($payload_json, false, 64, JSON_THROW_ON_ERROR);
523:         } catch (\JsonException $exception) {
524:             throw new InvalidTokenException('The JWT contains invalid JSON.', previous: $exception);
525:         }
526: 
527:         if (!$header instanceof \stdClass || !$payload instanceof \stdClass) {
528:             throw new InvalidTokenException('The JWT header and payload must be JSON objects.');
529:         }
530: 
531:         if (
532:             ($header->typ ?? null) !== 'JWT' || !isset($header->alg) || !is_string($header->alg)
533:             || property_exists($header, 'crit') || property_exists($header, 'b64')
534:         ) {
535:             throw new InvalidTokenException('The JWT header contains unsupported parameters.');
536:         }
537: 
538:         $key = $this->configuration['active_key'];
539: 
540:         if (property_exists($header, 'kid')) {
541:             if (!$this->isIdentifier($header->kid) || !array_key_exists($header->kid, $this->configuration['verification_keys'])) {
542:                 throw new InvalidTokenException('The JWT key ID is unknown.');
543:             }
544: 
545:             $key = $this->configuration['verification_keys'][$header->kid];
546:         } elseif ($this->configuration['key_id'] !== null) {
547:             throw new InvalidTokenException('The JWT key ID is missing.');
548:         }
549: 
550:         if ($header->alg !== $key['algorithm']) {
551:             throw new InvalidTokenException('The JWT algorithm does not match its configured key.');
552:         }
553: 
554:         $input = "$head.$body";
555:         $hash = 'sha' . substr($key['algorithm'], 2);
556: 
557:         if (str_starts_with($key['algorithm'], 'HS')) {
558:             if (!hash_equals(hash_hmac($hash, $input, $key['verification'], true), $signature)) {
559:                 throw new InvalidTokenException('The JWT signature is invalid.');
560:             }
561:         } else {
562:             $result = @openssl_verify($input, $signature, $key['verification'], $hash);
563: 
564:             if ($result === -1 || $result === false) {
565:                 throw new AuthSystemException('The JWT signature could not be checked.');
566:             }
567: 
568:             if ($result !== 1) {
569:                 throw new InvalidTokenException('The JWT signature is invalid.');
570:             }
571:         }
572: 
573:         // Preserve object-shaped roots while exposing nested claims as ordinary arrays.
574:         return json_decode($payload_json, true, 64, JSON_THROW_ON_ERROR);
575:     } // getJwtPayload
